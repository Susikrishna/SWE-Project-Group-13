\section{Summary}
% =============================================================================
\begin{frame}{Summary}
  \begin{block}{What we are building}
    A \textbf{centralized RBAC (Role-Based Access Control)} system to define roles,
    permissions, resource scoping, and dynamic policy evaluation.
  \end{block}
  \vspace{0.4cm}
  \begin{itemize}
    \item Flat, high-performance permission storage (no recursive role traversal)
    \item Support for \textbf{temporary roles} with expiry dates (delegation mechanism)
    \item \textbf{Microfrontend-aware} authorization --- the engine controls \emph{which UI modules}
          a user sees, not just which APIs they may call
    \item Centralized \textbf{Audit \& Registry} views for administrators
  \end{itemize}
\end{frame}

% =============================================================================
\section{Objectives \& Scope}
% =============================================================================
\begin{frame}{Objectives \& Scope}
  \begin{columns}[t]
    \column{0.48\textwidth}
    \begin{block}{Objective}
      Provide \textbf{secure, flexible permissioning} across all applications with
      centralized administration and audit capabilities.
    \end{block}
    \vspace{0.2cm}
    \begin{alertblock}{Key Design Goal}
      One authorization engine serves \emph{both} backend microservices and
      React microfrontend shells --- no duplicated permission logic.
    \end{alertblock}

    \column{0.48\textwidth}
    \begin{exampleblock}{Scope}
      \begin{itemize}
        \item Role definition \& storage
        \item Permission assignment per resource
        \item MFE access gating
        \item Role hierarchy (temporary roles)
        \item API Registry \& MFE Registry
        \item Admin consoles (Role Creator, Registry Manager)
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

\section{Functionalities Implemented}

% ── 4a: Authorization Engine ────────────────────────────────────────────────
\begin{frame}{Authorization Engine --- \texttt{auz-engine}}
  \begin{columns}[t]
    \column{0.55\textwidth}
    \textbf{Middleware pipeline} (every protected request):
    \begin{enumerate}
      \item \texttt{verifyToken} --- validates Bearer JWT, extracts \texttt{roleIds}
      \item \texttt{loadAccessProfile} --- fetches roles from DB, merges permissions
            \& MFE slugs into a flat set
      \item \texttt{requirePermission(key)} --- O(1) lookup, returns \texttt{403}
            if the key is absent
    \end{enumerate}
    \vspace{0.4cm}
    \textbf{REST endpoints}:
    \begin{itemize}
      \item \texttt{GET /auth/authorize} --- returns the user's full permission
            array \emph{and} the hydrated MFE objects (with \texttt{remoteUrl}/\texttt{module})
      \item \texttt{POST /auth/check-access} --- internal microservice call to
            verify a single \texttt{permissionKey}
    \end{itemize}

    \column{0.42\textwidth}
    \lstset{style=jsStyle, language=JavaScript}
    \begin{lstlisting}
// accessProfile.js (core logic)
function hasPermission(profile, key) {
  const k = key.trim().toLowerCase();
  return profile
    .mergedPermissions
    .includes(k); // O(1)
}

function isTempRoleCurrentlyValid(role) {
  if (!role.isTemp) return true;
  return new Date() <= new Date(role.expiresAt);
}
    \end{lstlisting}
  \end{columns}
\end{frame}

% ── 4b: Role Schema & Seed ──────────────────────────────────────────────────
\begin{frame}{Role Designer --- Schema \& Seeded Data}
  \begin{columns}[t]
    \column{0.50\textwidth}
    \textbf{Mongoose \texttt{Role} schema} (flattened, no joins):
    \begin{itemize}
      \item \texttt{\_id} --- custom string, e.g.\ \texttt{"role\_admin"}
      \item \texttt{permissions[]} --- e.g.\ \texttt{"user-service:user:read"}
      \item \texttt{mfeAccess[]} --- MFE feature slugs
      \item \texttt{isTemp} + \texttt{expiresAt} --- temporary delegation
    \end{itemize}
    \vspace{0.3cm}
    Permission string format:\\
    \texttt{\{service\}:\{resource\}:\{action\}}\\[4pt]
    e.g.\ \texttt{billing-service:invoice:create}

    \column{0.47\textwidth}
    \textbf{10 seeded roles}:

      \small support-agent       & Permanent \\
      \small finance-manager     & Permanent \\
      \small platform-admin      & Permanent \\
      \small analyst             & Permanent \\
      \small notification-manager & Permanent \\
      \small guest               & Permanent \\
      \small contractor-dev      & \textcolor{HighlightAmber}{Temporary} \\
      \small external-auditor    & \textcolor{HighlightAmber}{Temporary} \\
      \bottomrule

  \end{columns}
\end{frame}

% ── 4c: Registry Service ────────────────────────────────────────────────────
\begin{frame}{API \& MFE Registry Service}
  \begin{columns}[t]
    \column{0.48\textwidth}
    \begin{block}{API Registry}
      Stores every microservice endpoint with:
      \begin{itemize}
        \item \texttt{service}, \texttt{basePath}, \texttt{route}, \texttt{method}
        \item \texttt{resource}, \texttt{action}
        \item \texttt{permissionKey} (the canonical string used in roles)
        \item \texttt{isActive} toggle (soft decommission)
      \end{itemize}
      \medskip
      Seeded with \textbf{23 endpoints} across 5 services:
      auth, user, billing, notification, report.
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{MFE Registry}
      Stores every frontend module with:
      \begin{itemize}
        \item \texttt{feature} slug (linked to role \texttt{mfeAccess})
        \item \texttt{remoteUrl} --- Module Federation CDN/dev URL
        \item \texttt{module} --- Webpack exposed module name
        \item \texttt{route} --- shell router mount path
        \item \texttt{isActive} --- dark launch toggle
      \end{itemize}
      \medskip
      3 MFEs registered: \texttt{dashboard-mfe}, \texttt{admin-mfe}, \texttt{analytics-mfe}.
    \end{block}
  \end{columns}
\end{frame}

% ── 4d: Admin Registry Manager ─────────────────────────────────────────────
\begin{frame}{Admin Console --- Registry Management Portal}
  \begin{columns}[t]
    \column{0.55\textwidth}
    \textbf{Admin-Registry-Management} (Vite + React):
    \begin{itemize}
      \item Lists all registered \textbf{APIs and MFEs} in one unified view
      \item \textbf{Method badges} (GET/POST/PUT/DELETE) with colour coding
      \item \textbf{Live search} across service name, route, method, permission key,
            description --- results capped at 10, updated on every keystroke
      \item \textbf{Form pages} to register new microservices or microfrontends
      \item Loading / error / empty states handled gracefully
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Admin-Role-Creation} (Vite + React):
    \begin{itemize}
      \item \textbf{Role Designer} form --- create roles, assign permission strings
            and MFE slugs, set temporary expiry
      \item \textbf{Roles list} page --- view all stored roles
      \item Dashboard home with navigation
    \end{itemize}

    \column{0.42\textwidth}
    \begin{exampleblock}{Live Search Implementation}
      \lstset{style=jsStyle, language=JavaScript}

const filtered = useMemo(() => {
  if (!search.trim()) return allItems;
  const q = search.toLowerCase();
  return allItems
    .filter(item =>
      item._type === "API"
        ? [item.service, item.route,
           item.method,
           item.permissionKey]
            .some(v =>
              (v||"").toLowerCase()
              .includes(q))
        : [item.name, item.feature,
           item.route]
            .some(v =>
              (v||"").toLowerCase()

\end{frame}

    \node[step=HighlightAmber, right=1cm of rp]      (ctrl){Controller\\/ Service};

    \node[step=HighlightRed,   below=0.9cm of vt]    (r401){\texttt{401}\\Invalid JWT};
    \node[step=HighlightRed,   below=0.9cm of rp]    (r403){\texttt{403}\\Forbidden};

    \draw[arr] (client) -- node[above,font=\tiny]{Bearer JWT} (vt);
    \draw[arr] (vt)     -- (lap);
    \draw[arr] (lap)    -- (rp);
    \draw[arr] (rp)     -- (ctrl);
    \draw[arr,dashed,draw=HighlightRed] (vt)  -- (r401);
    \draw[arr,dashed,draw=HighlightRed] (rp)  -- (r403);
  \end{tikzpicture}
  \end{center}
  \vspace{0.2cm}
  \begin{itemize}
    \item Token payload carries \texttt{userId} and one or more \texttt{roleIds}
    \item Roles are fetched from MongoDB and \emph{merged} into a single permission set
    \item Temporary roles are validated against \texttt{expiresAt} \emph{before} merging
    \item A single \texttt{permissionKey} string (e.g.\ \texttt{user-service:user:delete})
          gates every endpoint
  \end{itemize}
\end{frame}

\section{What's Next}
% =============================================================================
\begin{frame}{Next Steps}
  \begin{columns}[t]
    \column{0.48\textwidth}
    \begin{block}{Access Simulator}
      A UI panel (in the Admin Console) that lets an administrator input a
      \texttt{userId} and a \texttt{permissionKey} and instantly see whether
      the user would be granted access --- without modifying any data.
    \end{block}
    \vspace{0.3cm}
    \begin{block}{Audit Logs}
      Append-only log collection capturing:
      \begin{itemize}
        \item Who modified which role and when
        \item Every \texttt{403} access denial (userId, resource, timestamp)
      \end{itemize}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{Attribute-Based (ABAC) Conditions}
      Extend the policy engine to support conditions such as:
      \begin{itemize}
        \item Time-of-day windows (``edit only during business hours'')
        \item IP / location constraints
        \item Resource-owner checks (\texttt{userId === resource.ownerId})
      \end{itemize}
    \end{block}
    \vspace{0.3cm}
    \begin{block}{Integration Testing}
      End-to-end tests covering the full JWT $\to$ permission gate flow using
      the demo-database seed data.
    \end{block}
  \end{columns}
\end{frame}
